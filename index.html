HÀM XỬ LÝ LÕI / コア機能の関数（かんすう） // ========================================== // Cập nhật giao diện màn hình điện tử / 液晶画面の表示更新（ひょうじこうしん） function updateDisplay() { const minutes = Math.floor(currentTime / 60); const seconds = currentTime % 60; const mm = String(minutes).padStart(2, '0'); const ss = String(seconds).padStart(2, '0'); timeDisplay.textContent = `${mm}:${ss}`; } // Logic tăng thời gian (+1 phút/+1 giây, hoặc +10 khi nhấn giữ, hoặc cộng dồn khi đang chạy) // 時間を増やすロジック（通常時、カウントダウン中、長押し時の挙動制御） function incrementTime(type, amount = 1) { if (isFlashing) return; // Chặn chỉnh thời gian khi đang nhấp nháy báo hiệu / 点滅中は操作無効 if (isRunning) { // A. KHI ĐANG ĐẾM NGƯỢC (Cộng thêm thời gian mà không dừng đếm) // カウントダウン中の時間追加（タイマーは止めない） if (type === 'min') { if (amount === 1) { currentTime += 60; // Thêm 1 phút (60 giây) / 1分（60秒）追加 } else { currentTime += amount * 60; // Nhấn giữ: thêm 10 phút / 長押し時：10分追加 } } else if (type === 'sec') { if (amount === 1) { currentTime += 10; // Thêm 10 giây / 10秒追加 } else { currentTime += amount; // Nhấn giữ: thêm 10 giây / 長押し時：10秒追加 } } } else { // B. KHI KHÔNG ĐẾM NGƯỢC (Trạng thái chờ hoặc tạm dừng - Giữ nguyên logic cũ) // 通常時（停止中・待機中）の時間設定（初期の仕様を維持） if (type === 'min') { currentTime += amount * 60; } else if (type === 'sec') { if (amount === 1) { let currentSec = currentTime % 60; let currentMin = Math.floor(currentTime / 60); currentSec = (currentSec + 1) % 60; currentTime = (currentMin * 60) + currentSec; } else { currentTime += amount; } } } // Giới hạn thời gian tối đa: 99 phút 59 giây (5999 giây) // 最大時間の制限（99分59秒 ＝ 5999秒） if (currentTime > 5999) { currentTime = 5999; } updateDisplay(); } // Cơ chế kích hoạt Bắt đầu / Tạm dừng / スタート・ストップの切り替え function toggleStartStop() { if (isFlashing) { stopFlashingAndRecall(); return; } if (isRunning) { // Tạm dừng bộ đếm / 一時停止（いちじていし） clearInterval(countdownInterval); isRunning = false; startStopLabel.textContent = 'START'; } else { // Kiểm tra điều kiện biên 00:00 / 00:00の時はスタートしない if (currentTime === 0) return; // Ghi nhớ mốc thời gian nếu bắt đầu chạy từ đầu / 初回スタート時に設定時間を記憶 if (initialSetTime === 0 || minBtn.disabled === false) { initialSetTime = currentTime; } // Bắt đầu đếm ngược / カウントダウン開始（かいし） isRunning = true; startStopLabel.textContent = 'STOP'; countdownInterval = setInterval(() => { if (currentTime > 0) { currentTime--; updateDisplay(); } else { // Khi chạm mức số 00:00 -> Kích hoạt báo hiệu / 00:00になったら点滅開始 clearInterval(countdownInterval); isRunning = false; startStopLabel.textContent = 'START'; triggerEndFlashing(); } }, 1000); } } // Thao tác Đặt lại toàn bộ (Reset) / タイマーのリセット処理 function resetTimer() { clearInterval(countdownInterval); clearTimeout(flashTimeout); clearInterval(flashInterval); currentTime = 0; initialSetTime = 0; isRunning = false; isFlashing = false; lcdDisplay.classList.remove('flash-active'); startStopLabel.textContent = 'START'; updateDisplay(); enableAdjustmentButtons(); } // Kích hoạt nhấp nháy màn hình LCD trong 3 giây khi kết thúc / 終了時の3秒間点滅処理 function triggerEndFlashing() { isFlashing = true; lcdDisplay.classList.add('flash-active'); disableAdjustmentButtons(); // Khóa nút bấm khi đang nhấp nháy báo hiệu / 点滅中のみボタンをロック // Tự động tắt nhấp nháy sau 3 giây và khôi phục thời gian cũ / 3秒後に自動停止してメモリから時間を復元 flashTimeout = setTimeout(() => { stopFlashingAndRecall(); }, 3000); } // Tắt nhấp nháy màn hình và khôi phục thời gian cài đặt trước đó / 点滅を止めて前回の設定時間に戻す function stopFlashingAndRecall() { clearTimeout(flashTimeout); lcdDisplay.classList.remove('flash-active'); isFlashing = false; currentTime = initialSetTime; // Trả về thời gian đặt trước đó / メモリーリコール updateDisplay(); enableAdjustmentButtons(); } // Khóa nút bấm / ボタンを無効化（むこうか）する function disableAdjustmentButtons() { minBtn.disabled = true; secBtn.disabled = true; } // Mở khóa nút bấm / ボタンを有効化（ゆうこうか）する function enableAdjustmentButtons() { minBtn.disabled = false; secBtn.disabled = false; } // ========================================== // 3. XỬ LÝ NHẤN GIỮ (LONG PRESS) & RESET CHUỘT // 長押し処理（ながおし）とマウスでのリセット検知 // ========================================== function startLongPressCounter(type) { if (isFlashing) return; // Thực hiện cộng phát đầu tiên / 1回目の増加 incrementTime(type, 1); // Thiết lập chờ sau 0.5s để kích hoạt chế độ nhảy liên tục (+10 đơn vị) // 0.5秒押し続けたら、0.15秒間隔で連続増加（+10ずつ）を開始 pressTimeout = setTimeout(() => { pressInterval = setInterval(() => { incrementTime(type, 10); }, LONG_PRESS_SPEED); }, LONG_PRESS_DELAY); } function stopLongPressCounter() { clearTimeout(pressTimeout); clearInterval(pressInterval); } // Kiểm tra Reset bằng chuột khi nhấp nhanh 2 nút trong vòng 0.3s // マウス操作用：0.3秒以内に両方のボタンがクリックされたらリセット function checkMouseReset(clickedType) { const now = Date.now(); if (clickedType === 'min') { lastMinClickTime = now; if (now - lastSecClickTime < MOUSE_RESET_WINDOW) { resetTimer(); } } else if (clickedType === 'sec') { lastSecClickTime = now; if (now - lastMinClickTime < MOUSE_RESET_WINDOW) { resetTimer(); } } } // ========================================== // 4. ĐĂNG KÝ SỰ KIỆN TƯƠNG TÁC / イベントリスナー // ========================================== // Sự kiện chuột cho nút MIN / 「分」ボタンのマウスイベント minBtn.addEventListener('mousedown', (e) => { if (e.button !== 0) return; // Chỉ nhận chuột trái / 左クリックのみ検知 checkMouseReset('min'); startLongPressCounter('min'); }); // Sự kiện chuột cho nút SEC / 「秒」ボタンのマウスイベント secBtn.addEventListener('mousedown', (e) => { if (e.button !== 0) return; checkMouseReset('sec'); startLongPressCounter('sec'); }); // Sự kiện thả chuột chung / マウスを離した時の処理 window.addEventListener('mouseup', stopLongPressCounter); // Hỗ trợ sự kiện cảm ứng trên Điện thoại (Smartphone) / スマホのタッチ操作対応 minBtn.addEventListener('touchstart', (e) => { e.preventDefault(); startLongPressCounter('min'); }); secBtn.addEventListener('touchstart', (e) => { e.preventDefault(); startLongPressCounter('sec'); }); window.addEventListener('touchend', stopLongPressCounter); // Sự kiện nút điều khiển Start/Stop / スタートボタンのクリックイベント startStopBtn.addEventListener('click', toggleStartStop); // ========================================== // 5. XỬ LÝ PHÍM TẮT BÀN PHÍM / キーボードショートカット // ========================================== window.addEventListener('keydown', (e) => { const key = e.key.toLowerCase(); // Tránh việc trình duyệt tự động lặp lại sự kiện keydown mặc định // キーの押しっぱなしによるブラウザ標準の連打イベントを防止 if (e.repeat && (key === 'q' || key === 'w')) return; activeKeys[key] = true; // CHẾ ĐỘ RESET: Nhấn đồng thời cả 2 phím Q và W // キーボードでのリセット：QとWの同時押し if (activeKeys['q'] && activeKeys['w']) { stopLongPressCounter(); resetTimer(); minBtn.classList.add('active-click'); secBtn.classList.add('active-click'); return; } // Phím Q - Tăng phút / Qキー：分追加 if (key === 'q' && !activeKeys['w']) { minBtn.classList.add('active-click'); startLongPressCounter('min'); } // Phím W - Tăng giây / Wキー：秒追加 if (key === 'w' && !activeKeys['q']) { secBtn.classList.add('active-click'); startLongPressCounter('sec'); } // Phím E - Khởi động / Dừng / Eキー：スタート・ストップ if (key === 'e') { startStopBtn.classList.add('active-click'); toggleStartStop(); } }); window.addEventListener('keyup', (e) => { const key = e.key.toLowerCase(); activeKeys[key] = false; if (key === 'q') { minBtn.classList.remove('active-click'); stopLongPressCounter(); } if (key === 'w') { secBtn.classList.remove('active-click'); stopLongPressCounter(); } if (key === 'e') { startStopBtn.classList.remove('active-click'); } }); // Khởi tạo hiển thị màn hình lúc ban đầu / 初期表示の実行 updateDisplay();
